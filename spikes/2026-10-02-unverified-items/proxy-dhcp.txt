# generated pxe/dnsmasq.conf (proxy mode)
# Generated by bin/pxe-lan. Do not edit; edit bin/pxe-lan.
interface=br0
bind-interfaces
# No DNS server.
port=0
# The whole DHCP/TFTP conversation per client goes to the log: "never reached
# iPXE" is distinguishable from "iPXE never chained".
log-dhcp
log-facility=/home/sqs/scripts/Windows-install-via-Linux/run/dnsmasq.log
# Our own lease file, not the system dnsmasq's.
dhcp-leasefile=/home/sqs/scripts/Windows-install-via-Linux/run/dnsmasq.leases
enable-tftp
tftp-root=/home/sqs/scripts/Windows-install-via-Linux/pxe
dhcp-match=set:efi64,option:client-arch,7
dhcp-match=set:efi64,option:client-arch,9
dhcp-match=set:ipxe,175
# Proxy-DHCP: answer only the PXE part; the LAN's DHCP server keeps handing out addresses.
# Every UEFI x86-64 PXE client gets OUR ipxe.efi, iPXE option ROMs included (see --bridge).
dhcp-range=10.42.0.0,proxy
pxe-service=tag:efi64,x86-64_EFI,"PDT network install",ipxe.efi
